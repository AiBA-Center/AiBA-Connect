⚙ ตั้งค่าเกมปิดท้ายการอบรม

ทำครั้งเดียว ใช้เวลาประมาณ 5 นาที ไม่ต้องสร้าง Google Form — แค่สร้าง Google Sheet 1 ไฟล์แล้ววางโค้ดที่เตรียมไว้ให้ ระบบจะสร้างตารางเก็บข้อมูลเองอัตโนมัติ

1

สร้าง Google Sheet ใหม่

ไปที่ sheets.new ตั้งชื่อไฟล์ เช่น เกมปิดท้ายการอบรม (ไม่ต้องพิมพ์อะไรในตาราง)
2

เปิดตัวเขียนสคริปต์

ในชีต กดเมนู ส่วนขยาย (Extensions) → Apps Script จะเปิดแท็บใหม่ที่มีโค้ดตัวอย่าง function myFunction() {} ให้ลบทิ้งทั้งหมด
3

วางโค้ดนี้แทน กดบันทึก (ไอคอนแผ่นดิสก์) แล้วทดสอบ

เคยวางโค้ดเวอร์ชันก่อนแล้ว? ต้องแทนด้วยโค้ดชุดนี้ (เวอร์ชัน 2 มีระบบให้แอดมินกดข้อถัดไป) แล้วกด Deploy → Manage deployments → ✏ แก้ไข → Version: New version → Deploy URL เดิมจะใช้ได้ต่อ ไม่ต้องสร้างลิงก์ใหม่
วิธีทดสอบที่ถูกต้อง: ที่แถบด้านบนของ Apps Script ตรงช่องเลือกฟังก์ชัน (ข้าง ▶ เรียกใช้) เลือก setup แล้วกด ▶ ครั้งแรก Google จะขอสิทธิ์ → อนุญาต → ด้านล่างจะขึ้น "พร้อมใช้งาน..." และในชีตจะมีแท็บ responses โผล่มา
ห้ามกด ▶ ที่ doPost หรือ doGet — สองฟังก์ชันนี้ Google เป็นคนเรียกตอนมีข้อมูลส่งมา ถ้ากดเองจะขึ้น error ซึ่งไม่ได้แปลว่าโค้ดผิด 
// ===== AiBA Connect — ตัวเก็บข้อมูลเกมปิดท้าย (เวอร์ชัน 2: แอดมินคุมข้อถัดไป) =====
// วิธีทดสอบ: เลือกฟังก์ชัน "setup" ด้านบน แล้วกด ▶ เรียกใช้ (ห้ามกดเรียกใช้ doPost/doGet ตรง ๆ)

var SHEET_NAME = 'responses';
var HEAD = ['เวลาบันทึก', 'ประเภท', 'ชื่อเล่น', 'คะแนน', 'เวลา(วินาที)', 'ข้อความ', 'รูป', 'รอบ/อารมณ์'];

function getSheet_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(SHEET_NAME);
  if (!sh) { sh = ss.insertSheet(SHEET_NAME); sh.appendRow(HEAD); sh.setFrozenRows(1); }
  return sh;
}
function jsonOut_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
function props_() { return PropertiesService.getScriptProperties(); }
function getState_() {
  var raw = props_().getProperty('STATE');
  var st = raw ? JSON.parse(raw) : { phase: 'lobby', qi: -1, startedAt: 0, round: 1 };
  st.now = Date.now();
  return st;
}
function setup() {
  var sh = getSheet_();
  if (!props_().getProperty('STATE')) props_().setProperty('STATE', JSON.stringify({ phase: 'lobby', qi: -1, startedAt: 0, round: 1 }));
  Logger.log('พร้อมใช้งาน ตาราง "' + SHEET_NAME + '" มีข้อมูล ' + Math.max(0, sh.getLastRow() - 1) + ' แถว');
}

function doPost(e) {
  try {
    if (!e || !e.postData) return jsonOut_({ ok: false, error: 'เรียกใช้โดยไม่มีข้อมูล (ให้ทดสอบด้วยฟังก์ชัน setup แทน)' });
    var d = JSON.parse(e.postData.contents || '{}');
    if (!d.type) return jsonOut_({ ok: false, error: 'no type' });

    // ---- คำสั่งจากแอดมิน (จอกลาง) ----
    if (d.type === 'control') {
      var saved = props_().getProperty('HOST_KEY');
      if (!saved) { props_().setProperty('HOST_KEY', String(d.key || '')); saved = String(d.key || ''); }
      if (!d.key || d.key !== saved) return jsonOut_({ ok: false, error: 'รหัสแอดมินไม่ถูกต้อง' });
      var st = getState_();
      if (d.action === 'start')      { st.phase = 'question'; st.qi = 0; st.startedAt = Date.now(); }
      else if (d.action === 'next')  { st.phase = 'question'; st.qi = (st.qi < 0 ? 0 : st.qi + 1); st.startedAt = Date.now(); }
      else if (d.action === 'end')   { st.phase = 'end'; }
      else if (d.action === 'reset') { st.phase = 'lobby'; st.qi = -1; st.startedAt = 0; st.round = (st.round || 1) + 1; }
      else if (d.action === 'lobby') { st.phase = 'lobby'; st.qi = -1; st.startedAt = 0; }
      delete st.now;
      props_().setProperty('STATE', JSON.stringify(st));
      return jsonOut_({ ok: true, state: getState_() });
    }

    // ---- ข้อมูลจากมือถือผู้เข้าอบรม ----
    var lock = LockService.getScriptLock(); lock.waitLock(5000);
    getSheet_().appendRow([new Date(), String(d.type), String(d.name || '').slice(0, 40), Number(d.score) || 0,
      Number(d.time) || 0, String(d.message || '').slice(0, 1000), String(d.photo || ''), String(d.mood || '')]);
    lock.releaseLock();
    return jsonOut_({ ok: true });
  } catch (err) { return jsonOut_({ ok: false, error: String(err) }); }
}

function doGet(e) {
  var p = (e && e.parameter) || {};
  if (p.action === 'state') return jsonOut_({ ok: true, state: getState_() });
  var sh = getSheet_();
  if (p.action === 'ping') return jsonOut_({ ok: true, rows: Math.max(0, sh.getLastRow() - 1), state: getState_() });
  var v = sh.getDataRange().getValues();
  var rows = [];
  for (var i = 1; i < v.length; i++) {
    var r = v[i];
    rows.push({ id: i - 1, ts: r[0], type: r[1], name: r[2], score: r[3], time: r[4], message: r[5], photo: r[6], mood: r[7] });
  }
  return jsonOut_({ ok: true, rows: rows, state: getState_() });
}
4

ทำให้ใช้งานได้ (Deploy)

กดปุ่ม ทำให้ใช้งานได้ → การทำให้ใช้งานได้ใหม่ (Deploy → New deployment) → กดไอคอนเฟือง เลือกประเภท เว็บแอป → ตั้งค่า:
  • ดำเนินการในฐานะ: ฉัน
  • ผู้ที่มีสิทธิ์เข้าถึง: ทุกคน (Anyone) ← สำคัญ ไม่งั้นมือถือผู้เข้าอบรมส่งข้อมูลไม่ได้
กด "ทำให้ใช้งานได้" → Google จะขอสิทธิ์ ให้กด อนุญาต (ถ้าเจอหน้า "ยังไม่ได้ยืนยันแอป" ให้กด ขั้นสูง → ไปที่... (ไม่ปลอดภัย) ซึ่งเป็นปกติสำหรับสคริปต์ที่เราเขียนเอง) → คัดลอก URL ของเว็บแอป ที่ลงท้ายด้วย /exec
ถ้าภายหลังแก้โค้ด ต้อง Deploy ใหม่เป็น "เวอร์ชันใหม่" ทุกครั้ง URL เดิมจึงจะอัปเดต

วาง URL แล้วสร้างลิงก์

✅ พร้อมใช้งาน

เปิดจอกลาง ↗
วันงาน: เปิดจอกลางบนโปรเจกเตอร์ → ฉาย QR ให้สแกน → ผู้เข้าอบรมใส่ชื่อเล่น+รูป แล้วกด "เข้าห้องควิซ" (จะเห็นหน้ารอ) → แอดมินกด ▶ เริ่มข้อ 1 บนจอกลาง ทุกเครื่องจะเห็นคำถามพร้อมกัน → กด ข้อถัดไป ทีละข้อ → กด จบเกม → 🏆 ประกาศผล
ลิงก์จอกลางมีรหัสแอดมินอยู่ในลิงก์ อย่าแชร์ให้ผู้เข้าอบรม
ล้างข้อมูลทดสอบ: ลบแถวในชีต responses (เก็บแถวหัวตารางไว้) หรือลบชีตทิ้ง ระบบจะสร้างใหม่เอง
